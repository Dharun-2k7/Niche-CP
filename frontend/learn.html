Track 01

Dynamic Programming (DP)

Dynamic Programming optimizes recursive solutions by storing intermediate results of subproblems to avoid redundant recalculations.

1. Memoization (Top-Down) vs Tabulation (Bottom-Up)

Memoization uses recursion with a cache array/hashmap to store computed state values. Tabulation builds states iteratively from base cases using loops.

C++ Python Go
// Fibonacci - Tabulation O(N) Time, O(N) Space int fib(int n) { if (n <= 1) return n; vector dp(n + 1, 0); dp[1] = 1; for (int i = 2; i <= n; i++) { dp[i] = dp[i - 1] + dp[i - 2]; } return dp[n]; }
# Fibonacci - Memoization O(N) Time memo = {} def fib(n): if n <= 1: return n if n not in memo: memo[n] = fib(n - 1) + fib(n - 2) return memo[n]
// Fibonacci - Bottom-Up O(N) Time func fib(n int) int { if n <= 1 { return n } dp := make([]int, n+1) dp[1] = 1 for i := 2; i <= n; i++ { dp[i] = dp[i-1] + dp[i-2] } return dp[n] }

2. 1D DP & Subsequence Problems (LIS)

In Longest Increasing Subsequence (LIS), state dp[i] represents the length of the longest strictly increasing subsequence ending at index i.

// LIS O(N^2) State Transition: dp[i] = max(dp[i], dp[j] + 1) for j < i and nums[j] < nums[i] int lengthOfLIS(vector& nums) { int n = nums.size(); if (n == 0) return 0; vector dp(n, 1); int ans = 1; for (int i = 1; i < n; i++) { for (int j = 0; j < i; j++) { if (nums[j] < nums[i]) { dp[i] = max(dp[i], dp[j] + 1); } } ans = max(ans, dp[i]); } return ans; }

3. 2D Grid DP & Matrix Paths

Find minimum path sum or total unique paths in an M x N grid moving only down or right:
dp[i][j] = grid[i][j] + min(dp[i-1][j], dp[i][j-1])

int minPathSum(vector>& grid) { int m = grid.size(), n = grid[0].size(); vector> dp(m, vector(n, 0)); dp[0][0] = grid[0][0]; for (int i = 1; i < m; i++) dp[i][0] = dp[i-1][0] + grid[i][0]; for (int j = 1; j < n; j++) dp[0][j] = dp[0][j-1] + grid[0][j]; for (int i = 1; i < m; i++) { for (int j = 1; j < n; j++) { dp[i][j] = grid[i][j] + min(dp[i-1][j], dp[i][j-1]); } } return dp[m-1][n-1]; }

4. 0/1 Knapsack & Unbounded Knapsack

Maximize profit given item weights and values within capacity W. Transition: dp[i][w] = max(dp[i-1][w], val[i-1] + dp[i-1][w - wt[i-1]]).

int knapsack(int W, vector& wt, vector& val, int n) { vector dp(W + 1, 0); for (int i = 0; i < n; i++) { for (int w = W; w >= wt[i]; w--) { dp[w] = max(dp[w], val[i] + dp[w - wt[i]]); } } return dp[W]; }

Breadth-First Search (BFS) & Shortest Path

BFS calculates unweighted shortest paths layer-by-layer using a FIFO queue.

void bfs(int start, vector>& adj, vector& visited) { queue q; q.push(start); visited[start] = true; while (!q.empty()) { int u = q.front(); q.pop(); for (int v : adj[u]) { if (!visited[v]) { visited[v] = true; q.push(v); } } } }

Disjoint Set Union (DSU) with Path Compression

struct DSU { vector parent, rank; DSU(int n) { parent.resize(n); iota(parent.begin(), parent.end(), 0); rank.assign(n, 0); } int find(int i) { if (parent[i] == i) return i; return parent[i] = find(parent[i]); } void unite(int i, int j) { int root_i = find(i), root_j = find(j); if (root_i != root_j) { if (rank[root_i] < rank[root_j]) swap(root_i, root_j); parent[root_j] = root_i; if (rank[root_i] == rank[root_j]) rank[root_i]++; } } };

Binary Exponentiation O(log N)

long long binpow(long long a, long long b, long long m) { a %= m; long long res = 1; while (b > 0) { if (b & 1) res = res * a % m; a = a * a % m; b >>= 1; } return res; }
